async function syncFromCloud() {
    const gistId = await getGistId();
    if(!gistId) { alert('请先在设置中配置Gist ID'); showPage('settings'); return; }
    
    showSyncStatus('🔄 同步中...', 'syncing');
    try {
        const r = await fetch('https://api.github.com/gists/' + gistId, {
            headers: { 'Authorization': 'token ' + GITHUB_TOKEN }
        });
        if(r.ok) {
            const data = await r.json();
            const files = Object.keys(data.files);
            if(files.length > 0) {
                let content = data.files[files[0]].content;
                // 尝试解析JSON
                try {
                    const json = JSON.parse(content);
                    if(json.workorders) {
                        workorders = json.workorders;
                        localStorage.setItem('workorders', JSON.stringify(workorders));
                        renderWorkorders();
                        renderDashboard();
                        showSyncStatus('✅ 同步成功 ' + workorders.length + '条', 'ok');
                        alert('同步成功！共' + workorders.length + '条工单');
                    } else {
                        alert('Gist内容不是工单数据格式，请先保存一次数据');
                    }
                } catch(e) {
                    // 内容不是JSON，可能是新建的Gist
                    alert('Gist是空的或格式不对，请先点击"保存到云端"创建数据');
                }
            }
        } else {
            showSyncStatus('❌ 同步失败', 'err');
            alert('同步失败，请检查Gist ID是否正确');
        }
    } catch(e) {
        showSyncStatus('❌ 同步失败', 'err');
        alert('同步失败: ' + e.message);
    }
}

async function uploadToCloud() {
    const gistId = await getGistId();
    if(!gistId) { alert('请先在设置中配置Gist ID'); showPage('settings'); return; }
    
    showSyncStatus('💾 保存中...', 'syncing');
    try {
        // 获取当前gist信息
        let sha = '';
        let fileName = 'workorders.json';
        try {
            const r = await fetch('https://api.github.com/gists/' + gistId, {
                headers: { 'Authorization': 'token ' + GITHUB_TOKEN }
            });
            if(r.ok) {
                const data = await r.json();
                sha = data.history[0].version;
                const files = Object.keys(data.files);
                if(files.length > 0) fileName = files[0];
            }
        } catch(e) {}
        
        const newContent = JSON.stringify({workorders: workorders, updateTime: new Date().toISOString()}, null, 2);
        
        const body = {
            description: '工单数据备份',
            public: false,
            files: {}
        };
        body.files[fileName] = { content: newContent };
        if(sha) body.files[fileName].sha = sha;
        
        const r = await fetch('https://api.github.com/gists/' + gistId, {
            method: 'PATCH',
            headers: { 
                'Authorization': 'token ' + GITHUB_TOKEN,
                'Content-Type': 'application/json'
            },
            body: JSON.stringify(body)
        });
        
        if(r.ok) {
            showSyncStatus('✅ 已保存', 'ok');
            alert('保存成功！');
        } else {
            showSyncStatus('❌ 保存失败', 'err');
            alert('保存失败');
        }
    } catch(e) {
        showSyncStatus('❌ 保存失败', 'err');
        alert('保存失败: ' + e.message);
    }
}
